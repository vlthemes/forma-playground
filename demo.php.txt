<?php
/**
 * Demo content of the WordPress Playground demo (blueprint.json runs it once, after the plugin is installed):
 * photos, portfolio items in four categories, ready Forma Layouts (Pro's too when Pro runs) and a page showing them.
 */

if ( !defined( 'ABSPATH' ) ) {
	exit;
}

use VLT\Forma\Core\Config;
use VLT\Forma\Core\PostTypes;

require_once ABSPATH . 'wp-admin/includes/image.php';

$forma_demo_dir = WP_CONTENT_DIR . '/forma-demo/photos/';
$forma_demo_pro = class_exists( '\VLT\Forma\Pro\Pro' );

// Photos into the Media Library
$forma_demo_images = [];

foreach ( glob( $forma_demo_dir . '*.webp' ) ?: [] as $file ) {
	$upload = wp_upload_bits( basename( $file ), null, (string) file_get_contents( $file ) );

	if ( !empty( $upload['error'] ) ) {
		continue;
	}

	$id = wp_insert_attachment( [
		'post_mime_type' => 'image/webp',
		'post_title'     => ucwords( str_replace( '-', ' ', basename( $file, '.webp' ) ) ),
		'post_status'    => 'inherit',
	], $upload['file'] );

	wp_update_attachment_metadata( $id, wp_generate_attachment_metadata( $id, $upload['file'] ) );
	$forma_demo_images[] = $id;
}

// Portfolio items: title, category, year, excerpt
$forma_demo_items = [

	[ 'Field Notes', 'Print', 2025, 'A field guide set in two colors, printed on uncoated stock.' ],

foreach ( $forma_demo_items as $k => $item ) {
	$term = term_exists( $item[1], 'portfolio_category' ) ?: wp_insert_term( $item[1], 'portfolio_category' );

	$id = wp_insert_post( [
		'post_type'    => 'portfolio',
		'post_status'  => 'publish',
		'post_title'   => $item[0],
		'post_excerpt' => $item[3],
		'post_content' => '<!-- wp:paragraph --><p>' . esc_html( $item[3] ) . '</p><!-- /wp:paragraph -->',
		'post_date'    => sprintf( '%d-%02d-%02d 10:00:00', $item[2], 12 - $k % 12, 1 + $k ),
	] );

	if ( $id && !is_wp_error( $term ) ) {
		wp_set_object_terms( $id, (int) $term['term_id'], 'portfolio_category' );
	}

	if ( $id && isset( $forma_demo_images[ $k ] ) ) {
		set_post_thumbnail( $id, $forma_demo_images[ $k ] );
	}
}

$forma_demo_shortcodes = '';

foreach ( $forma_demo_layouts as $title => $settings ) {
	$id = wp_insert_post( [
		'post_type'    => PostTypes::LAYOUT,
		'post_status'  => 'publish',
		'post_title'   => $title,
		// The editor's preview block (the post type's template is only filled in for new posts)
		'post_content' => '<!-- wp:forma/grid-settings /-->',
	] );

	if ( !$id || is_wp_error( $id ) ) {
		continue;
	}

	update_post_meta( $id, PostTypes::META_CONFIG, wp_slash( wp_json_encode( Config::sanitize( Config::merge( Config::defaults(), $settings ) ) ) ) );
	update_post_meta( $id, PostTypes::META_CSS, PostTypes::DEFAULT_CSS );

	$forma_demo_shortcodes .= sprintf( "<!-- wp:heading --><h2 class=\"wp-block-heading\">%s</h2><!-- /wp:heading -->\n<!-- wp:shortcode -->[forma_grid id=\"%d\"]<!-- /wp:shortcode -->\n", esc_html( $title ), $id );
}

// A page with every layout, to see them on the site
wp_insert_post( [
	'post_type'    => 'page',
	'post_status'  => 'publish',
	'post_title'   => 'Forma demo',
	'post_name'    => 'forma-demo',
	'post_content' => $forma_demo_shortcodes,
] );

// A demo opens where the blueprint says: no welcome redirect, no request for a review
delete_transient( 'forma_welcome_redirect' );
update_option( 'forma_review', 'done', false );

update_option( 'permalink_structure', '/%postname%/' );
flush_rewrite_rules();
